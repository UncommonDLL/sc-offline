cmake_minimum_required(VERSION 3.30)
project(launcher_safety_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(safety_tests transaction_tests.cpp ../../launcher/safety/transaction.cpp)
target_include_directories(safety_tests PRIVATE ../../launcher/safety)
add_test(NAME safety_tests COMMAND safety_tests)
add_executable(journal_tests journal_tests.cpp ../../launcher/safety/journal.cpp)
target_include_directories(journal_tests PRIVATE ../../launcher/safety)
target_link_libraries(journal_tests PRIVATE bcrypt)
add_test(NAME journal_tests COMMAND journal_tests)
add_executable(firewall_tests firewall_tests.cpp ../../launcher/safety/firewall.cpp ../../launcher/safety/journal.cpp)
target_include_directories(firewall_tests PRIVATE ../../launcher/safety)
target_link_libraries(firewall_tests PRIVATE bcrypt ole32 oleaut32 advapi32 ws2_32)
add_test(NAME firewall_tests COMMAND firewall_tests)
add_executable(recovery_tests recovery_tests.cpp ../../launcher/safety/windows_host.cpp ../../launcher/safety/transaction.cpp ../../launcher/safety/journal.cpp ../../launcher/safety/firewall.cpp)
target_include_directories(recovery_tests PRIVATE ../../launcher/safety)
target_link_libraries(recovery_tests PRIVATE bcrypt ole32 oleaut32 advapi32 ws2_32 shell32)
add_test(NAME recovery_tests COMMAND recovery_tests)
add_executable(windows_integration_tests windows_integration_tests.cpp ../../launcher/safety/journal.cpp ../../launcher/safety/firewall.cpp)
target_include_directories(windows_integration_tests PRIVATE ../../launcher/safety)
target_link_libraries(windows_integration_tests PRIVATE bcrypt ole32 oleaut32 advapi32 ws2_32)
